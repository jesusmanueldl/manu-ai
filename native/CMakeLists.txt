cmake_minimum_required(VERSION 3.22)
project(manu_jni)
set(LLAMA_ROOT "${CMAKE_CURRENT_LIST_DIR}/../third_party/llama.cpp" CACHE PATH "Ruta a llama.cpp")
add_library(manu_jni SHARED manu_jni.cpp)
target_include_directories(manu_jni PRIVATE "${LLAMA_ROOT}/include" "${LLAMA_ROOT}/ggml/include")
target_link_directories(manu_jni PRIVATE "${LLAMA_ROOT}/build-android/bin")
target_link_libraries(manu_jni llama log android)
set_target_properties(manu_jni PROPERTIES BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
